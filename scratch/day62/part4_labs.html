    python3 -c '
import time
import random

class TrueTimeSimulator:
    def __init__(self, epsilon_ms=5.0):
        self.epsilon = epsilon_ms / 1000.0  # uncertainty window in seconds
        
    def now(self):
        real_time = time.time()
        earliest = real_time - self.epsilon
        latest = real_time + self.epsilon
        return earliest, latest, self.epsilon

tt = TrueTimeSimulator(epsilon_ms=6.0)

# Simulate Transaction 1 (T1) committing
earliest_1, latest_1, eps_1 = tt.now()
commit_timestamp_1 = latest_1  # Assign commit timestamp >= t.latest
print(f"[T1] Assigned commit timestamp: {commit_timestamp_1:.6f}")
print(f"[T1] Uncertainty window: [{earliest_1:.6f}, {latest_1:.6f}] (eps={eps_1*1000:.1f}ms)")

# Enforce Commit Wait Rule: Hold locks until TrueTime.now().earliest > commit_timestamp_1
start_wait = time.time()
while True:
    cur_earliest, cur_latest, _ = tt.now()
    if cur_earliest > commit_timestamp_1:
        break
    time.sleep(0.001)

commit_wait_duration = time.time() - start_wait
print(f"[T1] Commit-wait completed in {commit_wait_duration*1000:.2f}ms (held locks until t > t.latest)")
print(f"[T1] Transaction committed and acknowledged to client.")

# Simulate Transaction 2 (T2) starting strictly after T1 confirmation
earliest_2, latest_2, eps_2 = tt.now()
commit_timestamp_2 = latest_2
print(f"[T2] Initiated after T1 acknowledgment.")
print(f"[T2] Assigned commit timestamp: {commit_timestamp_2:.6f}")

# Verify strict serializability invariant: commit_timestamp_2 > commit_timestamp_1
assert commit_timestamp_2 > commit_timestamp_1
delta_ms = (commit_timestamp_2 - commit_timestamp_1) * 1000.0
print(f"[INVARIANT VERIFIED] External Consistency Guaranteed: s2 > s1 by {delta_ms:.2f}ms without distributed read locks!")

  3. Model Bit-Reversed Primary Keys vs Sequential Hotspotting:

    python3 -c '
def bit_reverse_64(val):
    # Reverse bits of 64-bit integer
    b = bin(val)[2:].zfill(64)
    return int(b[::-1], 2)

num_splits = 16
split_size = (2**64) // num_splits

# Generate 1,000 sequential orders
sequential_keys = list(range(1, 1001))
bit_reversed_keys = [bit_reverse_64(k) for k in sequential_keys]

def calculate_split_distribution(keys):
    distribution = [0] * num_splits
    for k in keys:
        split_idx = min(k // split_size, num_splits - 1)
        distribution[split_idx] += 1
    return distribution

seq_dist = calculate_split_distribution(sequential_keys)
rev_dist = calculate_split_distribution(bit_reversed_keys)

print("Sequential Keys Distribution across 16 splits (HOTSPOT TRAP):")
print(f"  Split 0: {seq_dist[0]} inserts (100% on a single split!) | Splits 1-15: {seq_dist[1:]}")

print("\nBit-Reversed Keys Distribution across 16 splits (BALANCED FABRIC):")
for i, count in enumerate(rev_dist):
    print(f"  Split {i:02d}: {count} inserts")

# Assert uniform dispersion
assert all(50 <= count <= 80 for count in rev_dist)
print("\n[INVARIANT VERIFIED] Bit-reversal uniformly balances writes across 100% of Paxos splits.")

import threading
import time

class MockFirestoreDocument:
    def __init__(self, doc_id):
        self.doc_id = doc_id
        self.value = 1000
        self.version = 1
        self.lock = threading.Lock()
        self.aborts = 0
        self.commits = 0

    def update_transaction(self, decrement):
        with self.lock:
            current_version = self.version
            current_val = self.value
            
        time.sleep(0.005)  # simulate network roundtrip & transaction evaluation
        
        with self.lock:
            if self.version != current_version:
                self.aborts += 1
                raise Exception("FAILED_PRECONDITION: Aborted due to concurrent update (Paxos contention)")
            self.value = current_val - decrement
            self.version += 1
            self.commits += 1

doc = MockFirestoreDocument("inventory_summary")

def worker():
    for _ in range(25):
        try:
            doc.update_transaction(1)
        except Exception:
            pass

threads = [threading.Thread(target=worker) for _ in range(10)]
for t in threads: t.start()
for t in threads: t.join()

total_attempts = doc.commits + doc.aborts
print(f"Single Document Contention Results:")
print(f"  Total Attempts: {total_attempts}")
print(f"  Successful Commits: {doc.commits}")
print(f"  Contention Aborts:  {doc.aborts} ({doc.aborts/total_attempts*100:.1f}%)")

  2. Implement and Benchmark Distributed Sharded Counter Pattern:

    Execute a benchmark implementing a 20-shard distributed counter where writes are routed randomly across shards, and reads aggregate total stock using server-side SUM:

    python3 -c '
import random
import threading
import time

class DistributedShardedCounter:
    def __init__(self, num_shards=20):
        self.num_shards = num_shards
        self.shards = [0] * num_shards
        self.locks = [threading.Lock() for _ in range(num_shards)]
        self.total_writes = 0
        self.aborts = 0

    def increment(self, value=1):
        shard_id = random.randint(0, self.num_shards - 1)
        with self.locks[shard_id]:
            self.shards[shard_id] += value
            self.total_writes += 1

    def server_side_sum_aggregate(self):
        return sum(self.shards)

counter = DistributedShardedCounter(num_shards=20)

def sharded_worker():
    for _ in range(50):
        counter.increment(1)

threads = [threading.Thread(target=sharded_worker) for _ in range(20)]
start_time = time.time()
for t in threads: t.start()
for t in threads: t.join()
duration = time.time() - start_time

aggregated_total = counter.server_side_sum_aggregate()
print(f"Distributed 20-Shard Counter Results:")
print(f"  Total Operations Completed: {counter.total_writes} writes in {duration*1000:.1f}ms")
print(f"  Throughput: {counter.total_writes / duration:.0f} writes/sec")
print(f"  Contention Aborts: {counter.aborts} (0.0%)")
print(f"  Server-Side Aggregated Total: {aggregated_total}")
print(f"  Shard Distribution: {counter.shards}")

assert aggregated_total == 1000
print("[INVARIANT VERIFIED] Sharded counter eliminates document contention while preserving mathematical integrity.")
'

  1. Model Bigtable Lexicographical Ordering and Range Scan Behavior:

    Demonstrate how Bigtable's byte-order sorting behaves with raw sequential timestamps vs reverse timestamps:

    python3 -c '
import hashlib
import time

MAX_TIMESTAMP = 9999999999999

def generate_bad_key(device_id, ts):
    return f"{ts}#{device_id}"

def generate_production_key(tenant_id, device_id, ts):
    salt = hashlib.md5(device_id.encode()).hexdigest()[:4]
    reversed_ts = MAX_TIMESTAMP - ts
    return f"{salt}#{tenant_id}#{device_id}#{reversed_ts:013d}"

timestamps = [1700000001000, 1700000002000, 1700000003000, 1700000004000, 1700000005000]

bad_keys = [generate_bad_key("DEV-101", ts) for ts in timestamps]
prod_keys = [generate_production_key("TENANT-A", "DEV-101", ts) for ts in timestamps]

print("Standard Ascending Keys (Requires scanning to the tail to find latest event):")
for k in sorted(bad_keys):
    print(" ", k)

print("\nProduction Reverse Timestamp Keys (Latest event sorts FIRST lexicographically):")
for k in sorted(prod_keys):
    print(" ", k)

latest_prod_key = sorted(prod_keys)[0]
assert str(MAX_TIMESTAMP - 1700000005000) in latest_prod_key

def bad_key_tablet_mapper(key):

def evaluate_database_fit(workload_name, reqs):
    scores = {"Cloud Spanner": 0, "Cloud Firestore": 0, "Cloud Bigtable": 0, "Cloud SQL": 0}
    
    # 1. Transactional & Consistency Requirements
    if reqs.get("strict_serializability_multi_table"):
        scores["Cloud Spanner"] += 40
        scores["Cloud SQL"] += 20
        scores["Cloud Firestore"] -= 10
        scores["Cloud Bigtable"] -= 50
    elif reqs.get("document_atomic"):
        scores["Cloud Firestore"] += 35
        scores["Cloud Spanner"] += 10
    elif reqs.get("single_row_atomic_only"):
        scores["Cloud Bigtable"] += 30
        scores["Cloud Spanner"] += 10

    # 2. Scale & Concurrency
    qps = reqs.get("qps", 100)
    if qps > 100000:
        scores["Cloud Bigtable"] += 35
        scores["Cloud Spanner"] += 25
        scores["Cloud Firestore"] -= 20
        scores["Cloud SQL"] -= 50
    elif qps > 10000:
        scores["Cloud Spanner"] += 30
        scores["Cloud Bigtable"] += 20
        scores["Cloud Firestore"] += 10
        scores["Cloud SQL"] -= 10

    model = reqs.get("model")

        scores["Cloud Spanner"] += 30
        scores["Cloud SQL"] += 25

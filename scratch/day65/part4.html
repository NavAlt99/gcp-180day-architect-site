"""
Brightloaf Order Ingestion Integration Harness
Validates:
- OIDC ID token verification & IAM invoker authorization
- Rejection of missing/expired tokens (HTTP 401) and unauthorized callers (HTTP 403)
- Liveness (/healthz) vs Readiness (/readyz) connection pool decoupling
"""
import base64
import hashlib
import hmac
import json
import time

SECRET_KEY = b"brightloaf-gcp-mock-signing-key-32b"
SERVICE_URL = "https://order-service-xyz.us-central1.run.app"
AUTHORIZED_CALLERS = {"sa-api-gateway@brightloaf-prod.iam.gserviceaccount.com"}

def create_mock_jwt(email, audience, expired=False, invalid_sig=False):
    header = {"alg": "HS256", "typ": "JWT"}
    exp_time = int(time.time()) - 300 if expired else int(time.time()) + 3600
    payload = {
        "iss": "https://accounts.google.com",
        "sub": f"user-{hash(email) % 10000}",
        "email": email,
        "aud": audience,
        "exp": exp_time,
        "iat": int(time.time())
    }
    h_b64 = base64.urlsafe_b64encode(json.dumps(header).encode()).decode().rstrip("=")
    p_b64 = base64.urlsafe_b64encode(json.dumps(payload).encode()).decode().rstrip("=")
    signing_input = f"{h_b64}.{p_b64}".encode()
    key = b"wrong-key-bytes-for-test" if invalid_sig else SECRET_KEY
    sig = base64.urlsafe_b64encode(hmac.new(key, signing_input, hashlib.sha256).digest()).decode().rstrip("=")
    return f"{h_b64}.{p_b64}.{sig}"

def verify_oidc_bearer_token(auth_header):
    if not auth_header or not auth_header.startswith("Bearer "):
        return 401, {"error": "Missing or malformed Authorization header"}
    token = auth_header[7:].strip()
    parts = token.split(".")
    if len(parts) != 3:
        return 401, {"error": "Invalid JWT compact serialization"}
    h_b64, p_b64, sig_b64 = parts
    expected_sig = base64.urlsafe_b64encode(
        hmac.new(SECRET_KEY, f"{h_b64}.{p_b64}".encode(), hashlib.sha256).digest()
    ).decode().rstrip("=")
    if not hmac.compare_digest(sig_b64, expected_sig):
        return 401, {"error": "Cryptographic signature validation failed"}
    try:
        padding = "=" * (4 - len(p_b64) % 4) if len(p_b64) % 4 else ""
        payload = json.loads(base64.urlsafe_b64decode(p_b64 + padding))
    except Exception as e:
        return 401, {"error": f"Failed to deserialize claims: {e}"}
    if payload.get("exp", 0) < time.time():
        return 401, {"error": "Token has expired"}
    if payload.get("aud") != SERVICE_URL:
        return 403, {"error": f"Invalid audience claim: expected {SERVICE_URL}, got {payload.get('aud')}"}
    if payload.get("email") not in AUTHORIZED_CALLERS:
        return 403, {"error": f"Caller {payload.get('email')} lacks roles/run.invoker permission"}
    return 200, payload

class CloudSqlPscPool:
    def __init__(self, max_conns=4):
        self.max_conns = max_conns
        self.active_conns = 0
        self.endpoint_available = True
    def acquire(self):
        if not self.endpoint_available or self.active_conns >= self.max_conns:
            return None
        self.active_conns += 1
        return f"sock-psc-10.128.0.50:{5432 + self.active_conns}"
    def release(self):
        if self.active_conns > 0:
            self.active_conns -= 1

class OrderServiceServer:
    def __init__(self):
        self.pool = CloudSqlPscPool(max_conns=4)
        self.is_healthy = True
    def handle_request(self, method, path, headers, body=None):
        if path == "/healthz":
            if self.is_healthy:
                return 200, {"status": "SERVING", "uptime_sec": 128}
            return 500, {"status": "UNHEALTHY"}
        if path == "/readyz":
            sock = self.pool.acquire()
            if not sock:
                return 503, {"status": "UNAVAILABLE", "detail": "PSC connection pool exhausted or unhealthy"}
            self.pool.release()
            return 200, {"status": "READY", "psc_endpoint": "10.128.0.50:5432", "active_conns": self.pool.active_conns}
        if path == "/v1/orders":
            if method != "POST":
                return 405, {"error": "Method Not Allowed"}
            auth_header = headers.get("Authorization")
            status, auth_data = verify_oidc_bearer_token(auth_header)
            if status != 200:
                return status, auth_data
            sock = self.pool.acquire()
            if not sock:
                return 503, {"error": "Database unavailable: PSC pool exhausted"}
            try:
                order = json.loads(body) if body else {}
                return 201, {
                    "status": "CREATED",
                    "order_id": order.get("order_id", "ord-unknown"),
                    "principal": auth_data.get("email"),
                    "psc_socket": sock
                }
            finally:
                self.pool.release()
        return 404, {"error": "Not Found"}

Exercise 2: Distributed Trace Propagation and Inviolable Business Invariant Verification

Goal: Implement and verify W3C Trace Context (traceparent) and X-Request-Id propagation across the full request lifecycle, and execute a multi-threaded stress test demonstrating all four inviolable business invariants (atomicity, inventory check constraints, consumer deduplication, and monotonic state machines).

Mode: local exercise · Prerequisite: Day 64, Day 47, Day 54; bring their exit artifacts.

  1. Author the distributed invariant and trace test suite. Create verify_invariants_trace.py:

#!/usr/bin/env python3
"""
Brightloaf Distributed Trace & Business Invariants Harness
Verifies:
1. W3C Traceparent & X-Request-Id preservation across all hops
2. Invariant 1: Atomicity of Order and Outbox in single ACID transaction
3. Invariant 2: Inventory Integrity (CHECK stock >= 0) preventing overselling under concurrency
4. Invariant 3: Exactly-One Fulfillment via consumer deduplication table
5. Invariant 4: Monotonic State Progression (rejecting state regressions)
"""
import json
import sqlite3
import threading
import time
import uuid

# Connect to in-memory database
db = sqlite3.connect(":memory:", check_same_thread=False)
cursor = db.cursor()

# 1. Setup Relational Schema with hard invariants
cursor.executescript("""
CREATE TABLE inventory (
    item_id TEXT PRIMARY KEY,
    stock INTEGER NOT NULL,
    CONSTRAINT chk_stock_non_negative CHECK (stock >= 0)
);

CREATE TABLE orders (
    order_id TEXT PRIMARY KEY,
    customer_id TEXT NOT NULL,
    total REAL NOT NULL,
    status TEXT NOT NULL,
    traceparent TEXT NOT NULL,
    request_id TEXT NOT NULL,
    created_at TEXT NOT NULL
);

CREATE TABLE outbox_events (
    event_id TEXT PRIMARY KEY,
    order_id TEXT NOT NULL,
    payload TEXT NOT NULL,
    traceparent TEXT NOT NULL,
    request_id TEXT NOT NULL,
    published INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE processed_events (
    event_id TEXT PRIMARY KEY,
    order_id TEXT NOT NULL UNIQUE,
    processed_at TEXT NOT NULL
);

CREATE TABLE fulfillment_records (
    fulfillment_id TEXT PRIMARY KEY,
    order_id TEXT NOT NULL,
    carrier TEXT NOT NULL,
    tracking_number TEXT NOT NULL,
    dispatched_at TEXT NOT NULL
);

INSERT INTO inventory (item_id, stock) VALUES ('artisan-sourdough', 10);
""")
db.commit()

class OrderPipeline:
    def __init__(self, db_conn):
        self.db = db_conn
        self.lock = threading.Lock()
        self.pubsub_topic = [] # Simulated Cloud Pub/Sub topic

    def checkout(self, order_id, customer_id, item_id, quantity, total, traceparent, request_id, simulate_crash=False):
        c = self.db.cursor()
        try:
            with self.lock:
                c.execute("BEGIN TRANSACTION;")
                # Invariant 2: Decrement inventory protected by CHECK constraint
                c.execute("UPDATE inventory SET stock = stock - ? WHERE item_id = ?;", (quantity, item_id))
                
                # Invariant 1: Order insert
                c.execute("""
                    INSERT INTO orders (order_id, customer_id, total, status, traceparent, request_id, created_at)
                    VALUES (?, ?, ?, 'CREATED', ?, ?, datetime('now'));
                """, (order_id, customer_id, total, traceparent, request_id))
                
                if simulate_crash:
                    raise RuntimeError("Simulated crash before writing outbox event!")
                    
                # Invariant 1: Outbox insert in identical transaction
                event_id = f"evt-{order_id}-{uuid.uuid4().hex[:6]}"
                payload = json.dumps({"order_id": order_id, "item_id": item_id, "quantity": quantity})
                c.execute("""
                    INSERT INTO outbox_events (event_id, order_id, payload, traceparent, request_id)
                    VALUES (?, ?, ?, ?, ?);
                """, (event_id, order_id, payload, traceparent, request_id))
                
                c.execute("COMMIT;")
                
            # Asynchronous Outbox Relay to Pub/Sub
            self.pubsub_topic.append({
                "message_id": f"msg-{uuid.uuid4().hex[:8]}",
                "ordering_key": order_id,
                "data": payload,
                "attributes": {
                    "traceparent": traceparent,
                    "x-request-id": request_id,
                    "event_id": event_id
                }
            })
            return True, "ORDER_PLACED"
        except Exception as e:
            c.execute("ROLLBACK;")
            return False, str(e)

    def fulfill_consumer(self, message):
        """Asynchronous Fulfillment Consumer with Deduplication and Crash-Before-Ack"""
        attrs = message["attributes"]
        event_id = attrs["event_id"]
        traceparent = attrs["traceparent"]
        req_id = attrs["x-request-id"]
        payload = json.loads(message["data"])
        order_id = payload["order_id"]
        
        c = self.db.cursor()
        with self.lock:
            try:
                c.execute("BEGIN TRANSACTION;")
                
                # Invariant 3: Consumer Deduplication filter
                c.execute("""
                    INSERT OR IGNORE INTO processed_events (event_id, order_id, processed_at)
                    VALUES (?, ?, datetime('now'));
                """, (event_id, order_id))
                
                if c.rowcount == 0:
                    c.execute("ROLLBACK;")
                    return "DUPLICATE_DISCARDED", traceparent
                
                # Invariant 4: Monotonic State Machine Verification
                c.execute("SELECT status FROM orders WHERE order_id = ?;", (order_id,))
                row = c.fetchone()
                if not row:
                    c.execute("ROLLBACK;")
                    return "ORDER_NOT_FOUND", traceparent
                current_status = row[0]
                if current_status != "CREATED":
                    c.execute("ROLLBACK;")
                    return f"INVALID_TRANSITION_FROM_{current_status}", traceparent
                
                # Update status to FULFILLED
                c.execute("UPDATE orders SET status = 'FULFILLED' WHERE order_id = ?;", (order_id,))
                
                # Insert physical fulfillment dispatch record
                f_id = f"ful-{order_id}"
                c.execute("""
                    INSERT INTO fulfillment_records (fulfillment_id, order_id, carrier, tracking_number, dispatched_at)
                    VALUES (?, ?, 'USPS-Priority', ?, datetime('now'));
                """, (f_id, order_id, f"TRACK-{uuid.uuid4().hex[:10].upper()}"))
                
                # CRASH-BEFORE-ACK: Commit database state BEFORE returning ack
                c.execute("COMMIT;")
                return "FULFILLED", traceparent
            except Exception as e:
                c.execute("ROLLBACK;")
                return f"CONSUMER_ERROR: {e}", traceparent

    print("[PASS] Invariant 2 Verified: Database CHECK constraint prevented overselling.")

    cursor.execute("SELECT COUNT(*) FROM fulfillment_records WHERE order_id = 'ord-101';")
    assert cursor.fetchone()[0] == 1, "Fulfillment record count must remain exactly 1"
    print("[PASS] Invariant 3 Verified: Replayed Pub/Sub event caused zero duplicate dispatch.")

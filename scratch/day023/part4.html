    #!/usr/bin/env python3
"""Google Cloud Project Lifecycle & Lien Protection Engine.
Simulates state transitions: ACTIVE -> DELETE_REQUESTED (30-day recovery) -> DELETED.
Enforces programmatic project liens to block deletion requests.
"""
from __future__ import annotations
from typing import Dict, List, Optional

class ProjectLien:
    def __init__(self, origin: str, restriction: str, reason: str):
        self.origin = origin
        self.restriction = restriction
        self.reason = reason

class CloudProject:
    def __init__(self, project_id: str, project_number: int):
        self.project_id = project_id
        self.project_number = project_number
        self.state = "ACTIVE"
        self.billing_enabled = True
        self.services_active = True
        self.days_in_deletion = 0
        self.liens: Dict[str, ProjectLien] = {}

    def add_lien(self, origin: str, restriction: str, reason: str) -> None:
        self.liens[origin] = ProjectLien(origin, restriction, reason)
        print(f"[LIEN CREATED] Origin='{origin}', Restriction='{restriction}'")

    def remove_lien(self, origin: str) -> bool:
        if origin in self.liens:
            del self.liens[origin]
            print(f"[LIEN RELEASED] Origin='{origin}' removed.")
            return True
        return False

    def delete(self) -> None:
        print(f"\n>> Initiating projects.delete on '{self.project_id}'...")
        # Check active liens restricting deletion
        delete_liens = [l for l in self.liens.values() if l.restriction == "resourcemanager.projects.delete"]
        if delete_liens:
            active_reasons = "; ".join(l.reason for l in delete_liens)
            raise RuntimeError(
                f"HTTP 400 Bad Request / FAILED_PRECONDITION: "
                f"A lien is preventing '{self.project_id}' from being deleted: [{active_reasons}]"
            )
        
        # Transition to soft deletion
        self.state = "DELETE_REQUESTED"
        self.billing_enabled = False
        self.services_active = False
        self.days_in_deletion = 1
        print(f"[STATE TRANSITION] Project '{self.project_id}' is now DELETE_REQUESTED.")
        print("  - Compute instances halted immediately.")
        print("  - External static IP reservations released.")
        print("  - Cloud Billing automatically decoupled.")
        print("  - 30-day soft recovery window initiated.")

    def undelete(self) -> None:
        print(f"\n>> Initiating projects.undelete on '{self.project_id}'...")
        if self.state != "DELETE_REQUESTED":
            raise RuntimeError(f"Cannot undelete project in state '{self.state}'.")
        if self.days_in_deletion > 30:
            raise RuntimeError(f"Recovery window expired ({self.days_in_deletion} days). Project permanently purged.")

        self.state = "ACTIVE"
        self.billing_enabled = True  # Re-attached by administrator
        self.services_active = True
        self.days_in_deletion = 0
        print(f"[STATE TRANSITION] Project '{self.project_id}' restored to ACTIVE.")
        print("  - Metadata directory record restored.")
        print("  - Billing account re-linked successfully.")
        print("  - Services and APIs restored to operational state.")

def main():
    print("=" * 75)
    print("GOOGLE CLOUD PROJECT LIFECYCLE & LIEN SIMULATION")
    print("=" * 75)

    proj = CloudProject("bl-shared-artifacts-prod", 910284719283)
    print(f"Project Initialized: ID={proj.project_id}, State={proj.state}, Billing={proj.billing_enabled}")

    # Phase 1: Attach Protective Lien
    proj.add_lien(
        origin="terraform/core-landing-zone",
        restriction="resourcemanager.projects.delete",
        reason="Core shared artifact registry; accidental deletion will paralyze retail POS"
    )

    # Phase 2: Attempt Deletion While Lien Active
    try:
        proj.delete()
    except RuntimeError as err:
        print(f"EXPECTED CONTROL PLANE REJECTION:\n  {err}")

    # Phase 3: Release Lien and Execute Authorized Deletion
    print("\nAuthorized administrator releases lien for planned maintenance...")
    proj.remove_lien("terraform/core-landing-zone")
    proj.delete()

    print(f"\nCurrent Project State: {proj.state}, Billing Active: {proj.billing_enabled}")

    # Phase 4: Emergency Undelete During 30-Day Soft Recovery Window
    proj.undelete()
    print(f"\nFinal Project State: {proj.state}, Workloads Operational: {proj.services_active}")
    print("=" * 75)

if __name__ == "__main__":
    main()

    #!/usr/bin/env python3
"""Audits IAM Permission Bleed across Folder Hierarchy Topologies.
Compares Team-First hierarchy (shared folder) vs. Environment-First hierarchy.
"""
from __future__ import annotations
from typing import Dict, List, Set

class FolderNode:
    def __init__(self, name: str, parent: FolderNode | None = None):
        self.name = name
        self.parent = parent
        self.roles: Dict[str, List[str]] = {}
        self.projects: List[str] = []

    def assign_role(self, principal: str, role: str) -> None:
        self.roles.setdefault(principal, []).append(role)

    def get_effective_roles(self, principal: str) -> Set[str]:
        roles = set(self.roles.get(principal, []))
        if self.parent:
            roles.update(self.parent.get_effective_roles(principal))
        return roles

def main():
    print("=" * 75)
    print("ENTERPRISE HIERARCHY AUDIT: TEAM-FIRST VS. ENVIRONMENT-FIRST")
    print("=" * 75)

    principal = "group:mobile-developers@brightloaf.com"

    # MODEL A: Team-First Topology
    # Org -> /Mobile-App-Division -> [bl-mobile-staging, bl-mobile-prod]
    org_a = FolderNode("brightloaf.com (Org Root)")
    team_folder = FolderNode("Mobile-App-Division", parent=org_a)
    team_folder.assign_role(principal, "roles/editor")
    
    prod_a_roles = team_folder.get_effective_roles(principal)
    print("\n--- MODEL A: Team-First Folder Hierarchy ---")
    print("Structure: Org -> /Mobile-App-Division -> bl-mobile-prod & bl-mobile-staging")
    print(f"Folder Grant on /Mobile-App-Division: {team_folder.roles[principal]}")
    print(f"Effective Roles in 'bl-mobile-prod': {list(prod_a_roles)}")
    has_prod_write_a = "roles/editor" in prod_a_roles
    print(f">> VULNERABILITY ALERT: Developer write access in Production? {'YES (LEAKAGE DETECTED)' if has_prod_write_a else 'NO'}")

    # MODEL B: Environment-First Topology
    # Org -> /Production -> Mobile, /Non-Production -> Mobile
    org_b = FolderNode("brightloaf.com (Org Root)")
    prod_root = FolderNode("Production", parent=org_b)
    nonprod_root = FolderNode("Non-Production", parent=org_b)
    
    prod_mobile = FolderNode("Mobile-Prod", parent=prod_root)
    nonprod_mobile = FolderNode("Mobile-Staging", parent=nonprod_root)

    # Developers granted editor only in Non-Production branch
    nonprod_mobile.assign_role(principal, "roles/editor")
    prod_mobile.assign_role(principal, "roles/logging.viewer")

    staging_b_roles = nonprod_mobile.get_effective_roles(principal)
    prod_b_roles = prod_mobile.get_effective_roles(principal)

    print("\n--- MODEL B: Environment-First Folder Hierarchy ---")
    print("Structure: Org -> /Production -> bl-mobile-prod | /Non-Production -> bl-mobile-staging")
    print(f"Effective Roles in 'bl-mobile-staging': {list(staging_b_roles)}")
    print(f"Effective Roles in 'bl-mobile-prod':    {list(prod_b_roles)}")
    has_prod_write_b = "roles/editor" in prod_b_roles
    print(f">> SECURITY COMPLIANCE: Developer write access in Production? {'YES (FAIL)' if has_prod_write_b else 'NO (STRICT ISOLATION VERIFIED)'}")
    print("=" * 75)

if __name__ == "__main__":
    main()

    #!/usr/bin/env python3
"""Organization Policy Simulation, Test Plan & Rollback Engine.
Reviews a policy plan restricting locations (constraints/gcp.resourceLocations)
and external IPs (constraints/compute.vmExternalIpAccess).
Predicts accepted/rejected resources, executes admission simulation,
tests rollback capability, and generates org-policy-test-and-lifecycle-report.md.
"""
from __future__ import annotations
from pathlib import Path
from typing import Dict, List, Tuple

class OrgPolicyPlan:
    def __init__(self):
        self.enforce_no_external_ip = True
        self.allowed_locations = {"us-central1", "us-east1", "us-east4", "us-west1"}

    def evaluate(self, name: str, zone: str, has_external_ip: bool) -> Tuple[bool, str]:
        region = zone.rsplit("-", 1)[0]
        violations = []
        if self.enforce_no_external_ip and has_external_ip:
            violations.append("constraints/compute.vmExternalIpAccess (External IPs Denied)")
        if region not in self.allowed_locations:
            violations.append(f"constraints/gcp.resourceLocations (Region '{region}' not permitted in US locations)")

        if violations:
            return False, f"HTTP 400 FAILED_PRECONDITION: {'; '.join(violations)}"
        return True, "HTTP 200 OK: Resource provisioned successfully (Compliant with Org Policies)"

REPORT_TEMPLATE = """# Brightloaf Enterprise Cloud: Organization Policy & Lifecycle Report

## 1. Executive Summary & Core Architectural Invariant

def main():

    print("ORGANIZATION POLICY TEST PLAN & ADMISSION SIMULATION")
    print("=" * 75)

    policy = OrgPolicyPlan()

    test_cases = [
        ("TEST-01", "Backend Order API VM", "us-central1-a", False, "ACCEPTED"),
        ("TEST-02", "Analytics Worker VM",  "us-central1-a", True,  "REJECTED"),
        ("TEST-03", "Staging Test VM",      "europe-west3-b", False, "REJECTED"),
        ("TEST-04", "Contractor Test Node", "europe-west3-b", True,  "REJECTED"),
    ]

    print("\nRunning Simulated Policy Admission Checks:")
    all_passed = True
    for tid, desc, zone, ext_ip, pred in test_cases:
        ok, msg = policy.evaluate(desc, zone, ext_ip)
        obs = "ACCEPTED" if ok else "REJECTED"
        match = (pred == obs)
        all_passed = all_passed and match
        print(f"[{tid}] {desc} ({zone}, ext_ip={ext_ip}):")
